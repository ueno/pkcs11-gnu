# Intentional differences between p11-kit's pkcs11.h (CRYPTOKI_COMPAT
# mode) and pkcs11.h in this directory.  Format: <kind> <name pattern>.

# Internal helper macros of the old header.
macro CK_FUNCTION_LIST_*
macro _CK_DECLARE_FUNCTION

# Leaked from the old header: these GNU-to-standard renaming macros
# were not #undef'd at the end.
macro attribute_count
macro context_data_length
macro key_ptr
macro templ

# Leaked GNU typedefs.
typedef* ck_ec_kdf_t
typedef* ck_profile_id

# The old header used values that differ from the OASIS headers
# (0x1b5..0x1b9).
macro-value CKR_EXCEEDED_MAX_ITERATIONS
macro-value CKR_FIPS_SELF_TEST_FAILED
macro-value CKR_LIBRARY_LOAD_FAILED
macro-value CKR_PIN_TOO_WEAK
macro-value CKR_PUBLIC_KEY_INVALID

# Same value, type follows the standard headers ("1" in OASIS).
macro-type CKF_DONT_BLOCK
# Same value; the public domain header uses 0x00800000U instead of UL.
macro-type CKF_EC_NAMEDCURVE
